library(ggrepel)  # 用于避免标签重叠

# 设置分析范围 (0.9-1.1克拉)
diamonds_1ct <- diamonds %>%
  filter(between(carat, 0.9, 1.1)) %>%
  mutate(
    # 创建精确到百分位的克拉分组
    carat_centi = round(carat, 2),
    # 标记关键锚定点
    is_anchor = if_else(carat_centi == 1.00, "1.00克拉", "其他"),
    # 特别标记0.99克拉点
    is_99 = if_else(carat_centi == 0.99, "0.99克拉", "其他")
  )

# 计算每个0.01克拉分组的汇总数据
price_summary <- diamonds_1ct %>%
  group_by(carat_centi) %>%
  summarise(
    avg_price = mean(price),
    median_price = median(price),
    n = n(),
    .groups = "drop"
  ) %>%
  # 计算与1克拉的价格比率
  mutate(
    price_ratio = avg_price / avg_price[carat_centi == 1.00],
    price_ratio_label = scales::percent(price_ratio, accuracy = 1)
  )

# 计算0.99克拉与1.00克拉的价格差异
price_gap <- price_summary %>%
  filter(carat_centi %in% c(0.99, 1.00)) %>%
  select(carat_centi, avg_price) %>%
  pivot_wider(
    names_from = carat_centi,
    values_from = avg_price,
    names_prefix = "price_"
  ) %>%
  mutate(
    price_difference = price_1 - price_0.99,
    price_ratio = price_0.99 / price_1
  )

# 创建阶梯变化可视化
ggplot(price_summary, aes(x = carat_centi, y = avg_price)) +

  geom_text_repel(

  # 添加悬崖效应箭头
  geom_segment(

    color = "darkgreen", linetype = "dotted", size = 0.8
  ) +
  geom_segment(

# 设置分析区间和关键锚点
analysis_range <- c(0.9, 1.1)
anchor_points <- c(1, 1.5, 2)

# 数据预处理：过滤区间并创建锚定标签
diamonds_analysis <- diamonds %>%
  filter(between(carat, analysis_range[1], analysis_range[2])) %>%
  mutate(
    # 创建锚定分类：将钻石标记为整数锚点附近(±0.01)
    anchor_group = case_when(
      between(carat, 0.99, 1.01) ~ "1.0ct锚点区",
      between(carat, 1.49, 1.51) ~ "1.5ct锚点区",
      between(carat, 1.99, 2.01) ~ "2.0ct锚点区",
      TRUE ~ "非锚点区"
    ),
    # 创建精确到百分位的克拉分组
    carat_group = round(carat, 2)
  )

# 1. 价格曲线可视化（0.9-1.1克拉区间）
ggplot(diamonds_analysis, aes(carat, price)) +
  geom_point(aes(color = anchor_group), alpha = 0.3, size = 1.5) +
  geom_smooth(method = "loess", span = 0.2, color = "darkblue", se = FALSE) +
  geom_vline(xintercept = anchor_points, linetype = "dashed", color = "red") +
  labs(title = "钻石价格悬崖效应分析 (0.9-1.1克拉区间)",
       subtitle = "红色虚线表示消费者心理锚定点",
       x = "克拉重量", y = "价格(美元)",
       color = "关键锚定区域") +
  scale_x_continuous(breaks = seq(0.9, 1.1, 0.02)) +
  theme_minimal() +
  theme(legend.position = "top")

anchor_comparison <- diamonds_analysis %>%
  filter(anchor_group != "非锚点区") %>%
